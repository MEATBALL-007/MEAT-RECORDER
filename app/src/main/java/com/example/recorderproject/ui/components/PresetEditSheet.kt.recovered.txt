package com.example.recorderproject.ui.components

import androidx.compose.foundation.background
import androidx.compose.foundation.border
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.rememberScrollState
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.foundation.verticalScroll
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import com.example.recorderproject.model.ModeSettings
import com.example.recorderproject.model.RecordingMode

private val Orange = Color(0xFFE86A2B)
private val SheetBg = Color(0xFF111111)
private val RowBg   = Color(0xFF1A1A1A)

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun PresetEditSheet(
    mode: RecordingMode,
    currentSettings: ModeSettings,
    showLaunchButton: Boolean,
    onDismiss: (ModeSettings) -> Unit,
    onLaunch: (ModeSettings) -> Unit
) {
    var sampleRate       by remember(currentSettings) { mutableIntStateOf(currentSettings.sampleRate) }
    var bitDepth         by remember(currentSettings) { mutableIntStateOf(currentSettings.bitDepth) }
    var channelCount     by remember(currentSettings) { mutableIntStateOf(currentSettings.channelCount) }
    var noiseReduction   by remember(currentSettings) { mutableStateOf(currentSettings.noiseReduction) }
    var countdownSeconds by remember(currentSettings) {
        mutableIntStateOf(
            currentSettings.countdownSeconds.let { if (it in listOf(0, 3, 5)) it else 0 }
        )
    }
    var autoStopMinutes  by remember(currentSettings) {
        mutableIntStateOf(
            currentSettings.autoStopMinutes.let { if (it in listOf(0, 30, 60, 120)) it else 0 }
        )
    }

    fun snapshot() = ModeSettings(sampleRate, bitDepth, channelCount, noiseReduction, countdownSeconds, autoStopMinutes)

    ModalBottomSheet(
        onDismissRequest = { onDismiss(snapshot()) },
        sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true),
        containerColor = SheetBg,
        dragHandle = {
            Box(
                Modifier
                    .padding(top = 12.dp, bottom = 8.dp)
                    .width(40.dp)
                    .height(4.dp)
                    .clip(RoundedCornerShape(2.dp))
                    .background(Color(0xFF333333))
            )
        }
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 20.dp)
                .padding(bottom = 32.dp)
        ) {
            Text(
                text = "${mode.emoji}  ${mode.name}",
                color = Color(mode.accentArgb),
                fontSize = 14.sp,
                fontWeight = FontWeight.Bold,
                letterSpacing = 1.5.sp,
                modifier = Modifier.padding(bottom = 20.dp)
            )

            SheetLabel("SAMPLE RATE")
            PillSelector(
                options = listOf(44100, 48000, 96000),
                selected = sampleRate,
                label = { when (it) { 44100 -> "44.1 kHz"; 96000 -> "96 kHz"; else -> "48 kHz" } },
                onSelect = { sampleRate = it }
            )
            Spacer(Modifier.height(14.dp))

            SheetLabel("BIT DEPTH")
            PillSelector(
                options = listOf(16, 24, 32),
                selected = bitDepth,
                label = { when (it) { 32 -> "32-bit Float"; 24 -> "24-bit PCM"; else -> "16-bit" } },
                onSelect = { bitDepth = it }
            )
            Spacer(Modifier.height(14.dp))

            SheetLabel("CHANNELS")
            PillSelector(
                options = listOf(1, 2),
                selected = channelCount,
                label = { if (it == 1) "Mono" else "Stereo" },
                onSelect = { channelCount = it }
            )
            Spacer(Modifier.height(14.dp))

            SheetLabel("NOISE REDUCTION")
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                listOf(false to "Off", true to "On").forEach { (value, lbl) ->
                    PillChip(
                        label = lbl,
                        selected = noiseReduction == value,
                        modifier = Modifier.weight(1f),
                        onClick = { noiseReduction = value }
                    )
                }
            }
            Spacer(Modifier.height(14.dp))

            SheetLabel("COUNTDOWN")
            PillSelector(
                options = listOf(0, 3, 5),
                selected = countdownSeconds,
                label = { if (it == 0) "Off" else "${it}s" },
                onSelect = { countdownSeconds = it }
            )
            Spacer(Modifier.height(14.dp))

            SheetLabel("AUTO-STOP")
            PillSelector(
                options = listOf(0, 30, 60, 120),
                selected = autoStopMinutes,
                label = { when (it) { 0 -> "Off"; 30 -> "30 min"; 60 -> "1 hr"; else -> "2 hr" } },
                onSelect = { autoStopMinutes = it }
            )

            if (showLaunchButton) {
                Spacer(Modifier.height(24.dp))
                Button(
                    onClick = { onLaunch(snapshot()) }
